---------------------------

# Start Database
docker-compose up -d

# Stop Database
docker-compose down

# Check DB Status
docker ps

# Run the Application
go run main.go

# Format Code (Professional Standard)

4. DATABASE SCHEMA (SQL - Run inside Docker)
--------------------------------------------
-- Command to enter: docker exec -it radar-db-1 psql -U user -d momentum_db
CREATE EXTENSION IF NOT EXISTS postgis;

CREATE TABLE plans (
    id SERIAL PRIMARY KEY,
    title TEXT NOT NULL,
    location GEOGRAPHY(Point, 4326),
    base_score FLOAT DEFAULT 0,
    check_ins INT DEFAULT 0,
    saves INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

5. MOMENTUM FORMULA (The "Secret Sauce")
-----------------------------------------
Points = (Saves * 2.0) + (CheckIns * 3.0) + BaseScore
Gravity = 1.8
Score = Points / (TimeInHours + 2)^Gravity
